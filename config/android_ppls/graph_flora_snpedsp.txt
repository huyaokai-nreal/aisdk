  name: "HandDetectionTrack"                            # 节点名称

  input_stream: "IMAGE_INPUT:image"                     # 输入：图像

    input_stream_handler: "FixedSizeInputStreamHandler" # 固定大小输入处理器

        fixed_min_size: false                           # 不固定最小队列大小

  # calculator专用选项
  node_options: {

      model_name: "detect"                              # 使用的检测模型名称
